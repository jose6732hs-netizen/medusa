━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. INSTALAR DEPENDÊNCIAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   npm install

   Aguarde terminar (1-2 minutos)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. CONFIGURAR VARIÁVEIS DE AMBIENTE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   cp .env.example .env.local

   IMPORTANTE: Edite .env.local e configure:
   - DATABASE_URL = sua conexão Neon PostgreSQL


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. CRIAR TABELAS DO BANCO DE DADOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   npm run db:push

   Isso vai:
   ✓ Conectar ao Neon
   ✓ Criar 11 tabelas
   ✓ Preparar o banco para usar


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. RODAR O PROJETO LOCALMENTE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   npm run dev

   Vai abrir em: http://localhost:3000

   ✓ Hot Module Replacement ativo (mudanças aparecem ao vivo)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. FAZER PRIMEIRO ACESSO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   1. Abra http://localhost:3000 no navegador
   2. Clique em "Registrar"
   3. Crie sua conta (email e senha)
   4. Faça login
   5. Vá para Dashboard
   6. Crie sua primeira loja
   7. Adicione produtos
   8. Teste a storefront


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. FAZER DEPLOY (IR PARA O AR)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   OPÇÃO A: Deploy em Vercel (RECOMENDADO - Grátis)
   ─────────────────────────────────────────────────

   1. git push para seu repositório GitHub
   2. Acesse https://vercel.com/new
   3. Importe seu repositório
   4. Configure variáveis de ambiente
   5. Clique "Deploy"

   Vai ficar online automaticamente!


   OPÇÃO B: Deploy em Servidor Próprio
   ────────────────────────────────────

   1. npm run build
   2. npm start (inicia servidor de produção)

ESTRUTURA DE PASTAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   medusa-ecommerce/
   ├── app/                    # Páginas e API Routes
   ├── components/             # Componentes reutilizáveis
   ├── db/                     # Schema do banco
   ├── lib/                    # Utilities e helpers
   ├── scripts/                # Scripts de setup
   ├── public/                 # Arquivos estáticos
   └── package.json           # Dependências

   Pontos importantes:
   ✓ Cada página em app/ é uma rota
   ✓ Componentes em components/ podem ser reutilizados
   ✓ db/schema.ts define as tabelas
   ✓ API está em app/api/

PRECISA DE AJUDA?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Erros comuns:

   ❌ "Cannot find module '@neondatabase/serverless'"
   ✓ Solução: npm install

   ❌ "DATABASE_URL is not set"
   ✓ Solução: Configure DATABASE_URL no .env.local

   ❌ "Port 3000 is already in use"
   ✓ Solução: npm run dev -- -p 3001

   ❌ "Error connecting to database"
   ✓ Solução: Verifique sua DATABASE_URL do Neon
